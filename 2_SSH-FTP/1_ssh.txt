→ ssh korisnik@adresa      # povezivanje na udaljeni računar
→ pwd                      # prikaz radnog direktorijuma
→ ls                       # prikaz sadržaja radnog direktorijuma
→ mkdir direktorijum/      # kreiranje direktorijuma
→ cd direktorijum/         # prelazak u direktorijum
→ cd ../                   # povratak u naddirektorijum
→ rmdir direktorijum/      # brisanje praznog direktorijuma
→ touch fajl1              # kreiranje praznog fajla
→ nano fajl1               # otvaranje fajla u editoru; Ctrl + O, Enter, Ctrl + X
→ cat fajl1                # prikaz sadržaja fajla
→ cp fajl1 fajl2           # kopiranje fajla
→ mv fajl2 fajl3           # preimenovanje/premeštanje fajla
→ rm fajl3                 # brisanje fajla
→ clear                    # brisanje sadržaja terminala
→ history                  # prikaz istorije komandi
→ exit                     # prekid SSH konekcije
